const form = document.getElementById('propertyForm'); // Your form ID form.addEventListener('submit', async (e) => { e.preventDefault(); const formData = new FormData(form); // CRITICAL: Ensure image files are appended with the key 'photos' const fileInput = document.getElementById('photoInput'); // Your file input ID if (fileInput && fileInput.files.length > 0) { // If you manually append files, use 'photos' formData.delete('photos'); // Clear duplicate key if present for (let i = 0; i < fileInput.files.length; i++) { formData.append('photos', fileInput.files[i]); } } try { const response = await fetch('/api/submit', { method: 'POST', body: formData // DO NOT set Content-Type header manually when sending FormData }); const result = await response.json(); if (response.ok && result.success) { alert('Submitted successfully!'); form.reset(); } else { alert(`Error: ${result.error || 'Submission failed'}`); } } catch (err) { alert(`Error: ${err.message}`); } });
